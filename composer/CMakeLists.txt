cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
endif()
project(CrestComposer VERSION 0.2.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CREST_JUCE_SOURCE "" CACHE PATH "Optional local JUCE 8.0.12 checkout")
if(CREST_JUCE_SOURCE)
    add_subdirectory("${CREST_JUCE_SOURCE}" juce EXCLUDE_FROM_ALL)
    set(juce_root "${CREST_JUCE_SOURCE}")
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
    FetchContent_MakeAvailable(JUCE)
    set(juce_root "${juce_SOURCE_DIR}")
endif()

juce_add_plugin(CrestComposer COMPANY_NAME "Crest" BUNDLE_ID "com.crest.composer"
    PLUGIN_MANUFACTURER_CODE Crst PLUGIN_CODE CsK1 PRODUCT_NAME "Crest Composer"
    FORMATS VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE
    VST3_CATEGORIES Instrument Tools EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(CrestComposer PRIVATE Processor.cpp Editor.cpp EditorPages.cpp EditorLayout.cpp EditorFiles.cpp SketchGrid.cpp InstrumentStrip.cpp BoundKeys.cpp HarmonicAtlas.cpp Palette.cpp ColorRecipes.cpp BankBrowser.cpp LocalBank.cpp Sketch.cpp Audition.cpp GuitarBridge.cpp Performance.cpp Harmony.cpp)
target_include_directories(CrestComposer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_sources(CrestComposer PRIVATE TimelineSend.cpp)
target_compile_definitions(CrestComposer PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(CrestComposer PRIVATE juce::juce_audio_processors PUBLIC juce::juce_recommended_config_flags)

if(APPLE)
    set(bundle "$<TARGET_BUNDLE_DIR:CrestComposer_VST3>")
    add_custom_command(TARGET CrestComposer_VST3 POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${bundle}/Contents/Resources/Licenses"
        COMMAND ${CMAKE_COMMAND} -E copy "${juce_root}/LICENSE.md" "${bundle}/Contents/Resources/Licenses/JUCE.md"
        COMMAND ${CMAKE_COMMAND} -E copy "${juce_root}/modules/juce_audio_processors_headless/format_types/VST3_SDK/LICENSE.txt" "${bundle}/Contents/Resources/Licenses/VST3.txt"
        COMMAND /usr/bin/codesign --force --sign - "${bundle}" VERBATIM)
endif()

enable_testing()
foreach(domain harmony performance)
    string(SUBSTRING "${domain}" 0 1 initial)
    string(TOUPPER "${initial}" initial)
    string(SUBSTRING "${domain}" 1 -1 suffix)
    add_executable(${domain}_tests tests/${domain}.cpp ${initial}${suffix}.cpp)
    target_include_directories(${domain}_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME ${domain} COMMAND ${domain}_tests)
endforeach()
juce_add_console_app(guitar_tests PRODUCT_NAME "Composer guitar tests")
target_sources(guitar_tests PRIVATE GuitarBridge.cpp tests/guitar.cpp)
target_include_directories(guitar_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(guitar_tests PRIVATE juce::juce_core)
target_compile_definitions(guitar_tests PRIVATE JUCE_USE_CURL=0)
add_test(NAME guitar COMMAND guitar_tests)
find_program(node_exe node)
if(node_exe)
    add_test(NAME guitar-web COMMAND "${node_exe}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/guitar-web.mjs" $<TARGET_FILE:guitar_tests>)
endif()
juce_add_console_app(workflow_tests PRODUCT_NAME "Composer workflow tests")
target_sources(workflow_tests PRIVATE tests/workflow.cpp)
get_target_property(test_defs workflow_tests COMPILE_DEFINITIONS)
list(REMOVE_ITEM test_defs JUCE_STANDALONE_APPLICATION=1)
set_property(TARGET workflow_tests PROPERTY COMPILE_DEFINITIONS "${test_defs}")
target_compile_definitions(workflow_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(workflow_tests PRIVATE CrestComposer juce::juce_audio_processors)
add_test(NAME workflow COMMAND workflow_tests)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME timeline_placement COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_timeline.py")
endif()
